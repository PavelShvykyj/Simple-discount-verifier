Функция ОбработатьВебШтрихкод(Штрихкод, Заказ) Экспорт

    РезультатПроверкиФормата = ЭтоВебШтрихкод(Штрихкод);
    Если Не РезультатПроверкиФормата.Результативность Тогда
        Возврат РезультатПроверкиФормата;
    КонецЕсли;

    Штрихкод = РезультатПроверкиФормата.Данные;

    Если Не ЗначениеЗаполнено(Заказ) Тогда
        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Ложь,
            "Не вдалося визначити операцію продажу для перевірки знижки",
            Неопределено
        );
    КонецЕсли;

    РезультатПолученияПараметров = ПолучитьПараметрыСервисаПроверкиСкидок();
    Если Не РезультатПолученияПараметров.Результативность Тогда
        Возврат РезультатПолученияПараметров;
    КонецЕсли;

    ИдентификаторСканирования = Строка(Заказ.УникальныйИдентификатор());

    РезультатВалидации = ПроверитьШтрихкодЧерезСервис(
        Штрихкод,
        ИдентификаторСканирования
    );

    Если Не РезультатВалидации.Результативность Тогда
        ЗаписьЖурналаРегистрации(
            "ПроверкаСкидок.ПОС",
            УровеньЖурналаРегистрации.Ошибка,
            ,
            ,
            "Помилка перевірки веб-штрихкоду. идентификатор сканирования="
            + ИдентификаторСканирования
            + "; " + РезультатВалидации.Описание
        );

        Возврат РезультатВалидации;
    КонецЕсли;

    ДанныеВалидации = РезультатВалидации.Данные;

    Если Не ДанныеВалидации.Действителен Тогда
        ЗаписьЖурналаРегистрации(
            "ПроверкаСкидок.ПОС",
            УровеньЖурналаРегистрации.Информация,
            ,
            ,
            "Веб-штрихкод не прийнято. идентификатор сканирования="
            + ИдентификаторСканирования
            + "; причина=" + ДанныеВалидации.ПричинаОтказа
            + "; ідентифікатор зв'язку=" + ДанныеВалидации.ИдентификаторСвязи
        );

        РезультатТекстаОшибки = ТекстОшибкиВалидацииШтрихкода(
            ДанныеВалидации.ПричинаОтказа
        );
        Если Не РезультатТекстаОшибки.Результативность Тогда
            Возврат РезультатТекстаОшибки;
        КонецЕсли;

        ОписаниеОтказа = РезультатТекстаОшибки.Данные;
        Если ЗначениеЗаполнено(ДанныеВалидации.ИдентификаторСвязи) Тогда
            ОписаниеОтказа = ОписаниеОтказа
                + ". Код звернення: "
                + ДанныеВалидации.ИдентификаторСвязи;
        КонецЕсли;

        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Ложь,
            ОписаниеОтказа,
            ДанныеВалидации
        );
    КонецЕсли;

    РезультатНормализации = НормализоватьТелефонДляПоиска(
        ДанныеВалидации.КлючПоиска
    );
    Если Не РезультатНормализации.Результативность Тогда
        Возврат РезультатНормализации;
    КонецЕсли;

    Телефон = РезультатНормализации.Данные;

    РезультатПоискаКарты = НайтиДисконтнуюКартуПоТелефону(Телефон);
    Если Не РезультатПоискаКарты.Результативность Тогда
        Возврат РезультатПоискаКарты;
    КонецЕсли;

    ДанныеКарты = РезультатПоискаКарты.Данные;

    Если ЗначениеЗаполнено(ДанныеКарты.ДисконтнаяКарта) Тогда
        Если ДанныеКарты.Заблокирована Тогда
            ЗаписьЖурналаРегистрации(
                "ПроверкаСкидок.ПОС",
                УровеньЖурналаРегистрации.Информация,
                ,
                ,
                "Дисконтну картку заблоковано. идентификатор сканирования="
                + ИдентификаторСканирования
                + "; ідентифікатор зв'язку=" + ДанныеВалидации.ИдентификаторСвязи
            );

            Возврат Новый Структура(
                "Результативность,Описание,Данные",
                Ложь,
                "Дисконтну картку клієнта заблоковано",
                ДанныеКарты
            );
        КонецЕсли;

        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Истина,
            "Ок",
            ДанныеКарты.ДисконтнаяКарта
        );
    КонецЕсли;

    РезультатПолученияАнкеты = ПолучитьАнкетуЧерезСервис(
        ДанныеВалидации.КлючПоиска
    );

    Если Не РезультатПолученияАнкеты.Результативность Тогда
        ЗаписьЖурналаРегистрации(
            "ПроверкаСкидок.ПОС",
            УровеньЖурналаРегистрации.Ошибка,
            ,
            ,
            "Не вдалося отримати анкету клієнта. идентификатор сканирования="
            + ИдентификаторСканирования
            + "; ідентифікатор зв'язку=" + ДанныеВалидации.ИдентификаторСвязи
            + "; " + РезультатПолученияАнкеты.Описание
        );

        Возврат РезультатПолученияАнкеты;
    КонецЕсли;

    РезультатСозданияКарты = СоздатьЛокальныеДанныеКлиента(
        РезультатПолученияАнкеты.Данные,
        Телефон
    );

    Если Не РезультатСозданияКарты.Результативность Тогда
        ЗаписьЖурналаРегистрации(
            "ПроверкаСкидок.ПОС",
            УровеньЖурналаРегистрации.Ошибка,
            ,
            ,
            "Не вдалося створити локальні дані клієнта. идентификатор сканирования="
            + ИдентификаторСканирования
            + "; ідентифікатор зв'язку=" + ДанныеВалидации.ИдентификаторСвязи
            + "; " + РезультатСозданияКарты.Описание
        );

        Возврат РезультатСозданияКарты;
    КонецЕсли;

    Возврат Новый Структура(
        "Результативность,Описание,Данные",
        Истина,
        "Ок",
        РезультатСозданияКарты.Данные
    );

Функция НайтиДисконтнуюКартуПоТелефону(Телефон)

    ДанныеКарты = Новый Структура;
    ДанныеКарты.Вставить(
        "ДисконтнаяКарта",
        Справочники.ДисконтныеКарты.ПустаяСсылка()
    );
    ДанныеКарты.Вставить("Заблокирована", Ложь);

    Запрос = Новый Запрос;
    Запрос.Текст =
    "ВЫБРАТЬ ПЕРВЫЕ 1
    |   АнкетныеДанные.Владелец КАК ДисконтнаяКарта,
    |   АнкетныеДанные.Владелец.Заблокирована КАК Заблокирована
    |ИЗ
    |   Справочник.АнкетныеДанные КАК АнкетныеДанные
    |ГДЕ
    |   АнкетныеДанные.ПунктАнкеты = &ПунктТелефон
    |   И АнкетныеДанные.Значение = &Телефон";

    Запрос.УстановитьПараметр(
        "ПунктТелефон",
        глПараметрыСервисаПроверкиСкидок.ПунктАнкетыТелефон
    );
    Запрос.УстановитьПараметр("Телефон", Телефон);

    Попытка
        Выборка = Запрос.Выполнить().Выбрать();

        Если Выборка.Следующий() Тогда
            ДанныеКарты.ДисконтнаяКарта = Выборка.ДисконтнаяКарта;
            ДанныеКарты.Заблокирована = Выборка.Заблокирована;
        КонецЕсли;
    Исключение
        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Ложь,
            "Помилка пошуку дисконтної картки за телефоном: "
            + ОписаниеОшибки(),
            Неопределено
        );
    КонецПопытки;

    Возврат Новый Структура(
        "Результативность,Описание,Данные",
        Истина,
        "Ок",
        ДанныеКарты
    );

Функция СоздатьЛокальныеДанныеКлиента(Профиль, Телефон)

    РезультатПодготовкиАнкеты = ПодготовитьДанныеЗаполненияАнкеты(
        Профиль,
        Телефон
    );
    Если Не РезультатПодготовкиАнкеты.Результативность Тогда
        Возврат РезультатПодготовкиАнкеты;
    КонецЕсли;

    ДанныеЗаполненияАнкеты = РезультатПодготовкиАнкеты.Данные;
    ФИО = ДанныеЗаполненияАнкеты.Получить(
        глПараметрыСервисаПроверкиСкидок.ПунктАнкетыФИО
    );

    Если Не ЗначениеЗаполнено(ФИО) Тогда
        ФИО = Телефон;
    КонецЕсли;

    НачатьТранзакцию();

    Попытка
        Контрагент = Справочники.Контрагенты.СоздатьЭлемент();
        Контрагент.Родитель = Справочники.Контрагенты.Покупатели;
        Контрагент.Наименование = ФИО;
        Контрагент.Записать();

        ДисконтнаяКартаОбъект = Справочники.ДисконтныеКарты.СоздатьЭлемент();
        ДисконтнаяКартаОбъект.Наименование = ФИО;
        ДисконтнаяКартаОбъект.Выдана = ТекущаяДата();
        ДисконтнаяКартаОбъект.КодКарты = Телефон;
        ДисконтнаяКартаОбъект.РучнойКод = Телефон;
        ДисконтнаяКартаОбъект.Владелец = Контрагент.Ссылка;
        ДисконтнаяКартаОбъект.ПроцентСкидки = 10;
        ДисконтнаяКартаОбъект.Скидка = глПараметрыСервисаПроверкиСкидок.СкидкаПоУмолчанию;
        ДисконтнаяКартаОбъект.Видкарты = Справочники.ВидыДисконтныхКарт.Именная;
        ДисконтнаяКартаОбъект.ОбменДанными.Загрузка = Истина;
        ДисконтнаяКартаОбъект.Записать();

        ДисконтнаяКарта = ДисконтнаяКартаОбъект.Ссылка;

        ПланыВидовХарактеристик.ПунктыАнкетПотребителей.СоздатьПунктыПоВладельцу(
            ДисконтнаяКарта,
            Истина,
            ДанныеЗаполненияАнкеты
        );

        ЗафиксироватьТранзакцию();
    Исключение
        ОтменитьТранзакцию();

        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Ложь,
            "Помилка створення локальних даних клієнта: "
            + ОписаниеОшибки(),
            Неопределено
        );
    КонецПопытки;

    Возврат Новый Структура(
        "Результативность,Описание,Данные",
        Истина,
        "Ок",
        ДисконтнаяКарта
    );

Функция ПодготовитьДанныеЗаполненияАнкеты(Профиль, Телефон)

    ДанныеЗаполнения = Новый Соответствие;
    ДанныеЗаполнения.Вставить(
        глПараметрыСервисаПроверкиСкидок.ПунктАнкетыТелефон,
        Телефон
    );

    СоответствиеПунктовАнкеты =
        глПараметрыСервисаПроверкиСкидок.СоответствиеПунктовАнкеты;

    Ответы = Неопределено;

    Если Профиль <> Неопределено И Профиль.Свойство("answers", Ответы) Тогда
        Для Каждого Ответ Из Ответы Цикл
            КодОтвета = "";
            ЗначениеОтвета = "";

            Ответ.Свойство("code", КодОтвета);
            Ответ.Свойство("value", ЗначениеОтвета);

            ПунктАнкеты = СоответствиеПунктовАнкеты.Получить(КодОтвета);

            Если ЗначениеЗаполнено(ПунктАнкеты)
                И ЗначениеЗаполнено(ЗначениеОтвета) Тогда

                ДанныеЗаполнения.Вставить(ПунктАнкеты, ЗначениеОтвета);
            КонецЕсли;
        КонецЦикла;
    КонецЕсли;

    Возврат Новый Структура(
        "Результативность,Описание,Данные",
        Истина,
        "Ок",
        ДанныеЗаполнения
    );

Функция ПроверитьШтрихкодЧерезСервис(Штрихкод, ИдентификаторСканирования)

    ТелоЗапроса = Новый Структура;
    ТелоЗапроса.Вставить("barcodeValue", Штрихкод);
    ТелоЗапроса.Вставить(
        "terminalId",
        глПараметрыСервисаПроверкиСкидок.Терминал
    );
    ТелоЗапроса.Вставить(
        "branchId",
        глПараметрыСервисаПроверкиСкидок.Филиал
    );
    ТелоЗапроса.Вставить("scanId", ИдентификаторСканирования);

    РезультатЗапроса = ВыполнитьПостЗапросДжейсон(
        "/api/pos/barcodes/validate",
        ТелоЗапроса
    );
    Если Не РезультатЗапроса.Результативность Тогда
        Возврат РезультатЗапроса;
    КонецЕсли;

    ОтветСервера = РезультатЗапроса.Данные;

    Если ОтветСервера.КодСостояния <> 200 Тогда
        ОписаниеОтвета = "POST /api/pos/barcodes/validate: HTTP "
            + Строка(ОтветСервера.КодСостояния);
        Если ЗначениеЗаполнено(ОтветСервера.Тело) Тогда
            ОписаниеОтвета = ОписаниеОтвета + ". Ответ: " + ОтветСервера.Тело;
        КонецЕсли;

        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Ложь,
            ОписаниеОтвета,
            ОтветСервера
        );
    КонецЕсли;

    Попытка
        ДанныеОтвета = ОбработкаАнкет.СтрокаДжейсонВСтруктуру(
            ОтветСервера.Тело
        );
    Исключение
        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Ложь,
            "Не вдалося розібрати відповідь POST /api/pos/barcodes/validate: "
            + ОписаниеОшибки(),
            ОтветСервера
        );
    КонецПопытки;

    ДанныеВалидации = Новый Структура;
    ДанныеВалидации.Вставить("Действителен", Ложь);
    ДанныеВалидации.Вставить("ПричинаОтказа", "");
    ДанныеВалидации.Вставить("КлючПоиска", "");
    ДанныеВалидации.Вставить("ИдентификаторСвязи", "");
    ДанныеВалидации.Вставить("ПровереноВ", "");
    ДанныеВалидации.Вставить("ПовторнаяИдемпотентнаяПроверка", Ложь);

    Валидность = Ложь;
    Если Не ДанныеОтвета.Свойство("valid", Валидность) Тогда
        Возврат Новый Структура(
            "Результативность,Описание,Данные",
            Ложь,
            "Відповідь POST /api/pos/barcodes/validate не містить поле valid",
            ДанныеОтвета
        );
    КонецЕсли;

    ДанныеВалидации.Действителен = Валидность;

    Если Валидность Тогда
        КлючПоиска = "";
        ИдентификаторСвязи = "";
        ПровереноВ = "";
        ПовторнаяИдемпотентнаяПроверка = Ложь;

        Если Не ДанныеОтвета.Свойство("lookupKey", КлючПоиска)
            Или Не ЗначениеЗаполнено(КлючПоиска) Тогда

            Возврат Новый Структура(
                "Результативность,Описание,Данные",
                Ложь,
                "Відповідь успішної перевірки штрихкоду не містить lookupKey",
                ДанныеОтвета
            );
        КонецЕсли;

        Если Не ДанныеОтвета.Свойство("correlationId", ИдентификаторСвязи)
            Или Не ДанныеОтвета.Свойство("validatedAt", ПровереноВ)
            Или Не ДанныеОтвета.Свойство(
                "idempotentReplay",
                ПовторнаяИдемпотентнаяПроверка
            ) Тогда

            Возврат Новый Структура(
                "Результативность,Описание,Данные",
                Ложь,
                "Відповідь успішної перевірки штрихкоду не відповідає контракту АПІ",
                ДанныеОтвета
            );
        КонецЕсли;

        ДанныеВалидации.КлючПоиска = КлючПоиска;
        ДанныеВалидации.ИдентификаторСвязи = ИдентификаторСвязи;
        ДанныеВалидации.ПровереноВ = ПровереноВ;
        ДанныеВалидации.ПовторнаяИдемпотентнаяПроверка =
            ПовторнаяИдемпотентнаяПроверка;
    Иначе
        ПричинаОтказа = "unknown";
        ИдентификаторСвязи = "";
        ПровереноВ = "";

        Если Не ДанныеОтвета.Свойство("reason", ПричинаОтказа)
            Или Не ДанныеОтвета.Свойство("correlationId", ИдентификаторСвязи)
            Или Не ДанныеОтвета.Свойство("validatedAt", ПровереноВ) Тогда

            Возврат Новый Структура(
                "Результативность,Описание,Данные",
                Ложь,
                "Відповідь відхиленої перевірки штрихкоду не відповідає контракту АПІ",
                ДанныеОтвета
            );
        КонецЕсли;

        ДанныеВалидации.ПричинаОтказа = ПричинаОтказа;
        ДанныеВалидации.ИдентификаторСвязи = ИдентификаторСвязи;
        ДанныеВалидации.ПровереноВ = ПровереноВ;
    КонецЕсли;

    Возврат Новый Структура(
        "Результативность,Описание,Данные",
        Истина,
        "Ок",
        ДанныеВалидации
    );

Функция ПолучитьАнкетуЧерезСервис(Телефон)

    ТелоЗапроса = Новый Структура;
    ТелоЗапроса.Вставить("phone", Телефон);
    ТелоЗапроса.Вставить(
        "terminalId",
        глПараметрыСервисаПроверкиСкидок.Терминал

Функция СформироватьПодписьЗапроса(Метод, Путь, МоментЗапроса, ТелоДжейсон)

    РезультатХеширования = ПолучитьХешSHA256Строкой(ТелоДжейсон);
    Если Не РезультатХеширования.Результативность Тогда
        Возврат РезультатХеширования;
    КонецЕсли;

    РазделительСтрок = Символ(10);
    КаноническаяСтрока = ВРег(Метод);
    КаноническаяСтрока = КаноническаяСтрока + РазделительСтрок + Путь;
    КаноническаяСтрока = КаноническаяСтрока + РазделительСтрок + МоментЗапроса;
    КаноническаяСтрока = КаноническаяСтрока
        + РазделительСтрок
        + НРег(РезультатХеширования.Данные);

    Возврат ПолучитьПодписьHMACSHA256(КаноническаяСтрока);
